cmake_minimum_required(VERSION 3.16)
project(team04_host_driver_tests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(BSP_INC "${CMAKE_CURRENT_LIST_DIR}/Drivers/bsp/Inc")
set(BSP_SRC "${CMAKE_CURRENT_LIST_DIR}/Drivers/bsp/Src")
add_library(driver_core STATIC
    "${BSP_SRC}/lsm6ds3_c.c"
    "${BSP_SRC}/fifocapture_c.c"
    "${BSP_SRC}/sdcard_c.c")
target_include_directories(driver_core PUBLIC "${BSP_INC}")
target_compile_options(driver_core PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
foreach(test_name lsm6ds3_c fifocapture_c)
    add_executable(${test_name}_test tests/${test_name}_test.c)
    target_link_libraries(${test_name}_test PRIVATE driver_core)
    target_compile_options(${test_name}_test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME ${test_name}_test COMMAND ${test_name}_test)
endforeach()
add_executable(fifocapture_test
    "${BSP_SRC}/lsm6ds3.cpp" "${BSP_SRC}/fifocapture.cpp" tests/fifocapture_test.cpp)
target_link_libraries(fifocapture_test PRIVATE driver_core)
target_compile_options(fifocapture_test PRIVATE -Wall -Wextra -Werror -pedantic)
add_test(NAME fifocapture_test COMMAND fifocapture_test)
add_executable(sdcard_test "${BSP_SRC}/sdcard.cpp" tests/sdcard_test.cpp)
target_link_libraries(sdcard_test PRIVATE driver_core)
target_compile_options(sdcard_test PRIVATE -Wall -Wextra -Werror -pedantic)
add_test(NAME sdcard_test COMMAND sdcard_test)
